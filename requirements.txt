# ScanBox dependencies.
#
# Core - imported unconditionally; the app won't start without these.
wxPython
python-docx
fpdf2
browser-cookie3
Pillow

# Optional feature dependencies - each is wrapped in a try/except import in
# scanbox.py. The related feature becomes unavailable if missing, but all of
# these should be installed when building a fully featured end-user bundle.
pywin32; sys_platform == "win32"  # Windows scanner/WIA support.
numpy
opencv-python-headless   # NOT opencv-python - ScanBox never shows any cv2
                         # GUI window, and the headless build avoids extra
                         # Qt/GTK DLLs that PyInstaller-style builds can
                         # trip over.
pygrabber; sys_platform == "win32"  # Friendly DirectShow camera names.
winsdk; sys_platform == "win32"     # Native Windows.Media.Ocr screen reading.
onnxruntime              # Florence-2 local vision engine.
tokenizers               # Florence-2 local vision engine.
PyMuPDF                  # "fitz" import; PDF page reading.
accessible_output2; sys_platform == "win32"  # NVDA/JAWS/System Access/SAPI.
pywinauto; sys_platform == "win32"  # Read the active browser address without moving focus.
